# Edit > Replicate (replicate.oxtstack): the two padlock (link) buttons
# between the h and v fields of "Offset by" and "Scale by" show dark
# icons, made for the light appearance; on the dark window they were dim
# grey. In the dark appearance (revIDEAppearanceIsDark) they are tinted
# light with a colour overlay when the dialog opens, as the IDE tints its
# other icon buttons (revcorestackbehavior); in the light one the overlay
# is cleared, so they look as they did. An IDE without the function (a
# script that opens the stack on its own) keeps them as they were.
file: Toolset/palettes/replicate.oxtstack
object: card 1
replace:
|  set the loc of stack "revReplicate" to the loc of the topStack|
|  unlock messages|
|end preOpenStack|
with:
|  set the loc of stack "revReplicate" to the loc of the topStack|
|  revReplicateLockIconColours|
|  unlock messages|
|end preOpenStack|
||
|-- OXT-Beyond: the padlocks' icons are tinted light in the dark appearance|
|on revReplicateLockIconColours|
|  local tDark, tError, tLock|
|  put false into tDark|
|  try|
|    put revIDEAppearanceIsDark() into tDark|
|  catch tError|
|  end try|
|  repeat for each item tLock in "offsetLockButton,scaleLockButton"|
|    if tDark is true then|
|      set the colorOverlay["color"] of btn tLock of me to "210,210,210"|
|      set the colorOverlay["opacity"] of btn tLock of me to 255|
|    else|
|      set the colorOverlay of btn tLock of me to empty|
|    end if|
|  end repeat|
|end revReplicateLockIconColours|
