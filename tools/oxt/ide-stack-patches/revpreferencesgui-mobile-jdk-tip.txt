# Preferences > Mobile Support: with no JDK found, the tip over the JDK
# path named Java 8 ("please install the 'jdk-8u202-windows.exe' directly",
# and the same for the Mac's and Linux's), OpenXTalk Lite's advice. The
# Android SDK's tools that the builder runs no longer start on Java 8 (d8
# of build-tools 35 stops with UnsupportedClassVersionError, #353), and
# OXT-Beyond's Android builds are made and tested with JDK 17, so the tip
# sent people to a JDK that cannot build. It names JDK 17 now, on every
# platform, and the button beside the path that chooses a JDK's folder.
# One line, where the Mac's old two did not fit the tip's 36 pixels.
# Text lines are between bars, so that trailing blanks survive editors.
file: Toolset/palettes/revpreferencesgui.rev
object: card "Mobile Support"
replace:
|   if the platform contains "mac" and tJDK is empty then|
|      -- no JDK installed on Mac|
|      put "For MacOS JDK support, please install the 'jdk-8u202-macosx-x64.dmg' directly." into cd fld "fmobtip"|
|      show cd fld "fmobtip"|
|   end if|
|   if the platform contains "win" and tJDK is empty  then|
|      -- no JDK installed on windows|
|      put "For Windows JDK support, please install the 'jdk-8u202-windows.exe' directly." into cd fld "fmobtip"|
|      show cd fld "fmobtip"|
|   end if|
|   if the platform contains "lin" and tJDK is empty  then|
|      put "For Linux JDK support, please select the 'jdk-8u202' JDK path manually." into cd fld "fmobtip"|
|      show cd fld "fmobtip"|
|   end if|
