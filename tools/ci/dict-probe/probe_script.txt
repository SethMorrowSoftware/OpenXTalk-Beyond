local sLog, sStarted
on openStack
   if sStarted then pass openStack
   put true into sStarted
   insert the script of this stack into front
   send "probe" to this stack in 15 seconds
   pass openStack
end openStack

on errorDialog pErr
   plog "ERRORDIALOG " & line 1 to 6 of pErr
   pass errorDialog
end errorDialog

on plog pText
   put the long time && pText & return after sLog
   put sLog into URL ("file:" & $OXT_PROBE_OUT)
end plog

on probe
   if $OXT_PROBE_DARK is "1" then
      set the cIDEAppearance of stack "revPreferences" to "dark"
      try
         revIDEApplyAppearance
      catch tErr
         plog "THROW appearance " & line 1 of tErr
      end try
      plog "ide dark=" & revIDEIsDark()
   end if
   plog "system appearance=" & the systemAppearance & " app=" & the appAppearance
   plog "platform=" & the platform & " version=" & the version & " engine=" & specialFolderPath("engine")
   plog "prefs display=" & the cDictionaryDisplayMethod of stack "revPreferences" & " quick=" & the cQuickDictionary of stack "revPreferences"
   -- the text dictionary
   set the cDictionaryDisplayMethod of stack "revPreferences" to true
   set the cQuickDictionary of stack "revPreferences" to false
   try
      send "tNewDictionaryDisplay" to stack "revMenubar"
   catch tErr
      plog "THROW text " & line 1 to 4 of tErr
   end try
   wait 4 seconds with messages
   plog "text dict open=" & (there is a stack "OpenXTalk Lite Dictionary")
   if there is a stack "OpenXTalk Lite Dictionary" then
      plog "text dict mode=" & the mode of stack "OpenXTalk Lite Dictionary" & " vis=" & the visible of stack "OpenXTalk Lite Dictionary"
      plog "text dict path=" & field "path" of card 1 of stack "OpenXTalk Lite Dictionary"
      plog "text dict entries=" & the number of lines of field "tEntries" of card 1 of stack "OpenXTalk Lite Dictionary"
      plog "text dict entry=" & line 1 to 3 of field "styledTextField" of card 1 of stack "OpenXTalk Lite Dictionary"
      try
         put "answer" into field "searchFld" of card 1 of stack "OpenXTalk Lite Dictionary"
         send "textChanged" to field "searchFld" of card 1 of stack "OpenXTalk Lite Dictionary"
         wait 2 seconds with messages
         plog "text dict search lines=" & the number of lines of field "tDropSearches" of card 1 of stack "OpenXTalk Lite Dictionary" & " first=" & line 1 of field "tDropSearches" of card 1 of stack "OpenXTalk Lite Dictionary"
      catch tErr
         plog "THROW text search " & line 1 to 4 of tErr
      end try
      export snapshot from rect (the rect of stack "OpenXTalk Lite Dictionary") to file ($OXT_PROBE_OUT & "-text.png") as PNG
      close stack "OpenXTalk Lite Dictionary"
   end if
   -- the Quick Dictionary
   set the cQuickDictionary of stack "revPreferences" to true
   try
      send "tNewDictionaryDisplay" to stack "revMenubar"
   catch tErr
      plog "THROW quick " & line 1 to 4 of tErr
   end try
   wait 4 seconds with messages
   plog "quick open=" & (there is a stack "Quick Dictionary")
   if there is a stack "Quick Dictionary" then
      plog "quick mode=" & the mode of stack "Quick Dictionary" & " conn=" & the cConnID of card 1 of stack "Quick Dictionary"
      plog "quick bg=" & the effective backgroundColor of stack "Quick Dictionary"
      try
         plog "drivers=" & revGetDatabaseDriverPath()
      catch tErr
         plog "THROW drivers " & line 1 of tErr
      end try
      try
         put "answer" into field "Cerca2" of card 1 of stack "Quick Dictionary"
         send "SearchDB" to field "Cerca2" of card 1 of stack "Quick Dictionary"
         wait 1 second with messages
         plog "quick search lines=" & the number of lines of the dgText of group "risultati" of card 1 of stack "Quick Dictionary" & " first=" & line 1 of the dgText of group "risultati" of card 1 of stack "Quick Dictionary"
         plog "quick sqlite=" & revDataFromQuery(tab, cr, the cConnID of card 1 of stack "Quick Dictionary", "SELECT sqlite_version()")
         plog "quick count=" & revDataFromQuery(tab, cr, the cConnID of card 1 of stack "Quick Dictionary", "SELECT count(*) FROM dictionary_data")
      catch tErr
         plog "THROW quick search " & line 1 to 4 of tErr
      end try
      export snapshot from rect (the rect of stack "Quick Dictionary") to file ($OXT_PROBE_OUT & "-quick.png") as PNG
   end if
   set the cQuickDictionary of stack "revPreferences" to false
   plog "openstacks=" & the openStacks
   plog "done"
   quit
end probe
